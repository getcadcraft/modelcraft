ລູກບາດມຸມມອງຢູ່ມຸມເທິງຂວາຂອງມຸມມອງ 3D. ມັນຫມຸນຕາມມຸມມອງ, ດັ່ງນັ້ນມັນຈະສະແດງສະເໝີວ່າທ່ານກຳລັງເບິ່ງໄປທາງໃດ: {{ui:ເທິງ}} ຢູ່ດ້ານເທິງ, {{ui:ໜ້າ}} ຫັນໄປທາງໃຕ້ (ທ່ານເບິ່ງໄປທາງເໜືອ, ຕາມແກນສີຂຽວ, ເມື່ອທ່ານເບິ່ງມັນ), {{ui:ຂວາ}} ຫັນໄປທາງຕາເວັນອອກ (ແກນສີແດງ). ມັນຈະໂປ່ງແສງຈົນກວ່າທ່ານຈະຊີ້ໃສ່ມັນ.

## ເບິ່ງຈາກດ້ານ, ຂອບ ຫຼືມຸມ

1. ຊີ້ໃສ່ລູກບາດ. ສ່ວນທີ່ຢູ່ໃຕ້ຕົວຊີ້ຈະສະຫວ່າງຂຶ້ນ: ກາງໜ້າ, ຂອບ (ມັນຈະສະຫວ່າງທັງສອງໜ້າທີ່ໃຊ້ຮ່ວມກັນ) ຫຼືມຸມ (ໃນທັງສາມໜ້າ).
2. ຄລິກມັນ. ມຸມມອງຈະໝຸນເພື່ອເບິ່ງຈາກບ່ອນນັ້ນ, ໂດຍຮັກສາຈຸດກາງ ແລະການຊູມເທົ່າເດີມ.

ມີ 26 ຈຸດໃຫ້ຄລິກ: 6 ໜ້າ ({{ui:ເທິງ}}, {{ui:ລຸ່ມ}}, {{ui:ໜ້າ}}, {{ui:ຫຼັງ}}, {{ui:ຊ້າຍ}}, {{ui:ຂວາ}}), 12 ຂອບ ແລະ 8 ມຸມ. ມຸມຈະໃຫ້ມຸມມອງສາມສ່ວນສີ່ແບບຄລາສສິກ; ຂອບຈະເບິ່ງຢູ່ກາງລະຫວ່າງສອງໜ້າ.

ການຄລິກ {{ui:ໜ້າ}} ຄືກັນກັບ {{ui:ມຸມມອງ}}, {{ui:ໜ້າ}} ໃນແຖບເຄື່ອງມື, ແຕ່ບໍ່ຊູມໃຫ້ພໍດີ. {{ui:ມຸມມອງ}} ຍັງຊູມໃຫ້ເຫັນທັງໂມເດວ.

## ໝຸນໂດຍການລາກລູກບາດ

ລາກລູກບາດເພື່ອໝຸນມຸມມອງ, ຄືກັນກັບການໝຸນໃນມຸມມອງເອງ. ການລາກຈະບໍ່ຖືກນັບເປັນການຄລິກ, ດັ່ງນັ້ນທ່ານສາມາດຈັບລູກບາດຢູ່ບ່ອນໃດກໍໄດ້.

## ວົງເຂັມທິດ

ວົງທີ່ຢູ່ໃຕ້ລູກບາດມີ N, E, S ແລະ W.

- **ລາກວົງ** ເພື່ອໝຸນໂມເດວຮອບແກນຕັ້ງ, ຄືກັບວ່າມັນຢູ່ເທິງແທ່ນໝຸນ. ວົງຈະຕາມຕົວຊີ້.
- **ຄລິກຕົວອັກສອນ** ເພື່ອເບິ່ງຈາກດ້ານນັ້ນ, ໂດຍຮັກສາມຸມທີ່ທ່ານເບິ່ງລົງ. ໃນມຸມມອງຈາກເທິງລົງມາຕົງໆ, ຕົວອັກສອນທີ່ທ່ານຄລິກຈະໄປຢູ່ດ້ານລຸ່ມຂອງໜ້າຈໍ.

## ມຸມມອງໜ້າຕົງ: ລູກສອນ

ເມື່ອທ່ານເບິ່ງໜ້າຕົງ, ລູກສອນນ້ອຍຈະປາກົດຮອບລູກບາດ:

- ສາມຫຼ່ຽມ**ສີ່ອັນ** ນຳໜ້າຢູ່ດ້ານນັ້ນມາໄວ້ດ້ານໜ້າ: ສາມຫຼ່ຽມເທິງລູກບາດສະແດງໜ້າທີ່ເຄີຍຢູ່ດ້ານເທິງ, ແລະອື່ນໆ.
- ສອງ**ລູກສອນໂຄ້ງ** ທີ່ມຸມເທິງຂວາຈະໝຸນພາບຄັ້ງລະໜຶ່ງສ່ວນສີ່ທາງເຂັມໂມງ ຫຼືທວນເຂັມໂມງ, ໂດຍບໍ່ປ່ຽນໜ້າທີ່ທ່ານກຳລັງເບິ່ງ. ໃຊ້ມັນເພື່ອໝຸນແຜນຜັງ ເພື່ອໃຫ້ຝາຍາວຢູ່ຂ້າມໜ້າຈໍ.

## ກັບສູ່ມຸມມອງເລີ່ມຕົ້ນ

ປຸ່ມເຮືອນທີ່ມຸມເທິງຊ້າຍຂອງລູກບາດຈະກັບໄປຫາມຸມມອງສາມສ່ວນສີ່ {{ui:ໄອໂຊ}} ແລະປັບໃຫ້ໂມເດວທັງໝົດພໍດີ. {{key:Home}} ເຮັດແບບດຽວກັນໃນຂະນະທີ່ລູກບາດມີໂຟກັສຄີບອດ.

## ດ້ວຍແປ້ນພິມ

ກົດ {{key:Tab}} ເພື່ອໄປຫາລູກບາດ, ແລ້ວ:

| ປຸ່ມ | ເຮັດ |
|---|---|
| {{key:left}} / {{key:right}} | ເລືອກດ້ານ, ຂອບ ຫຼືມຸມຮອບລູກບາດ, ເປັນຂັ້ນລະ 1/8 ຮອບ (ໂປຣແກຣມອ່ານໜ້າຈໍຈະບອກວ່າແມ່ນອັນໃດ) |
| {{key:up}} / {{key:down}} | ເລືອກອັນທີ່ສູງກວ່າ ຫຼືຕ່ຳກວ່າ, ຂຶ້ນໄປຫາ {{ui:ເທິງ}} ຫຼືລົງໄປຫາ {{ui:ລຸ່ມ}} |
| {{key:Enter}} ຫຼື {{key:Space}} | ເບິ່ງຈາກອັນທີ່ເລືອກ |
| {{key:Shift+left}} / {{key:Shift+right}} | ໝຸນມຸມມອງໜຶ່ງສ່ວນສີ່ຮອບແກນຕັ້ງ |
| {{key:Home}} | ມຸມມອງເລີ່ມຕົ້ນ |
| {{key:Esc}} | ອອກຈາກລູກບາດ |

ປຸ່ມທີ່ກົດໃສ່ລູກບາດຈະຢູ່ກັບລູກບາດເທົ່ານັ້ນ: ພວກມັນຈະບໍ່ເປີດເຄື່ອງມື ModelCraft.

## ໄອຄອນແກນ ຫຼືບໍ່ມີຫຍັງແທນ

ເປີດ {{ui:ການຕັ້ງຄ່າ}} ({{key:Mod+,}}), ແທັບ {{ui:ModelCraft}}, ແລ້ວຕັ້ງ **ຕົວຊີ້ມຸມ** ເປັນ {{ui:ລູກບາດມຸມມອງ}} (ຄ່າເລີ່ມຕົ້ນ), {{ui:ໄອຄອນແກນ}} (ເຄື່ອງໝາຍ X, Y, Z ນ້ອຍຢູ່ມຸມລຸ່ມຊ້າຍ: X ສີແດງ, Y ສີຂຽວ, Z ສີຟ້າ) ຫຼື {{ui:ບໍ່ມີ}}. ບຣາວເຊີນີ້ຈະຈື່ການເລືອກ. ເບິ່ງ [ການຕັ້ງຄ່າ ModelCraft](topic:settings#corner-indicator).

:::note
ModelCraft ບໍ່ມີການໝຸນແນວຕັ້ງ (roll): ນອກຈາກມຸມມອງລົງຕົງ ແລະຂຶ້ນຕົງ, ດ້ານເທິງຂອງໜ້າຈໍຈະເປັນດ້ານເທິງຂອງໂມເດວສະເໝີ. ລູກບາດ ແລະລູກສອນໂຄ້ງສາມາດໝຸນມຸມມອງແຜນຜັງໄດ້; ພວກມັນບໍ່ສາມາດເອນມຸມມອງດ້ານຂ້າງໃຫ້ລົ້ມມາຢູ່ຂອບໄດ້.
:::
